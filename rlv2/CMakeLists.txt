# Prefer building from the repository root CMakeLists.txt (sets DRONE_RL_DIR automatically).
cmake_minimum_required(VERSION 3.20)
project(rlv2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)

# drone-rl checkout, used only by the parity reference.
set(DRONE_RL_DIR "" CACHE PATH "Path to a drone-rl checkout")

if(DRONE_RL_DIR)
    add_executable(rlv2_reference reference.cpp)
    target_include_directories(rlv2_reference PRIVATE ${DRONE_RL_DIR}/include)
    target_compile_options(rlv2_reference PRIVATE -O2 -Wall -Wextra)

    find_package(Threads REQUIRED)
    find_package(BLAS)
    if(BLAS_FOUND OR APPLE)
        add_executable(rlv2_grade grade.cpp)
        target_include_directories(rlv2_grade PRIVATE ${DRONE_RL_DIR}/include)
        target_compile_options(rlv2_grade PRIVATE -O2 -Wall -Wextra)
        target_link_libraries(rlv2_grade PRIVATE Threads::Threads ${BLAS_LIBRARIES})
        if(APPLE)
            target_link_libraries(rlv2_grade PRIVATE "-framework Accelerate")
            target_compile_definitions(rlv2_grade PRIVATE ACCELERATE_NEW_LAPACK)
        endif()
    endif()

    add_executable(rlv2_controller_study controller_study.cpp)
    target_include_directories(rlv2_controller_study PRIVATE ${DRONE_RL_DIR}/include)
    target_compile_options(rlv2_controller_study PRIVATE -O2 -Wall -Wextra)
endif()

# Teacher training on CPU threads, for smoke tests on machines without Metal.
find_package(Threads REQUIRED)
find_package(BLAS)
if(BLAS_FOUND OR APPLE)
    add_executable(rlv2_train_cpu train_cpu.cpp)
    target_include_directories(rlv2_train_cpu PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/cpu)
    target_compile_options(rlv2_train_cpu PRIVATE -O2 -Wall -Wno-attributes -Wno-unused-variable)
    target_link_libraries(rlv2_train_cpu PRIVATE Threads::Threads ${BLAS_LIBRARIES})
    if(APPLE)
        target_link_libraries(rlv2_train_cpu PRIVATE "-framework Accelerate")
        target_compile_definitions(rlv2_train_cpu PRIVATE ACCELERATE_NEW_LAPACK)
    endif()
endif()

if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 17)
    get_filename_component(REPO_ROOT ${CMAKE_CURRENT_SOURCE_DIR} DIRECTORY)
    add_executable(rlv2 rlv2.mm)
    target_compile_options(rlv2 PRIVATE -O3 -fobjc-arc -Wall -Wextra)
    target_compile_definitions(rlv2 PRIVATE SOURCE_DIR="${REPO_ROOT}")
    target_link_libraries(rlv2 PRIVATE "-framework Foundation" "-framework Metal")

    add_executable(rlv2_train train.mm)
    target_compile_options(rlv2_train PRIVATE -O3 -fobjc-arc -Wall -Wextra -Wno-unused-parameter)
    target_compile_definitions(rlv2_train PRIVATE SOURCE_DIR="${REPO_ROOT}" ACCELERATE_NEW_LAPACK)
    target_link_libraries(rlv2_train PRIVATE "-framework Foundation" "-framework Metal" "-framework Accelerate")
endif()
